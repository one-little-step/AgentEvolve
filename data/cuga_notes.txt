import numba
@numba.njit
def lev_numba(a,b):
    n=len(a); m=len(b)
    prev=np.zeros(m+1,dtype=np.int32)
    cur=np.zeros(m+1,dtype=np.int32)
    for j in range(m+1):
        prev[j]=j
    for i in range(1,n+1):
        cur[0]=i
        ai=a[i-1]
        for j in range(1,m+1):
            cost=0 if ai==b[j-1] else 1
            cur[j]=min(prev[j]+1, cur[j-1]+1, prev[j-1]+cost)
        tmp=prev.copy()
        prev[:]=cur[:]
        cur[:]=tmp[:]
    return prev[m]
