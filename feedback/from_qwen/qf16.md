class StorageBackend(Protocol):
    def write_record(self, record_type: str, record_id: str,
                     payload: Mapping[str, object]) -> None: ...
    def read_record(self, record_type: str, record_id: str
                    ) -> Mapping[str, object] | None: ...
    def list_records(self, record_type: str
                     ) -> tuple[Mapping[str, object], ...]: ...
    def close(self) -> None: ...

class JSONFileStorage:
    """Simple JSON-file storage for research runs.
    
    Future work: replace with SQLite WAL for production deployment.
    """
    def __init__(self, root: Path) -> None:
        self._root = root
        self._root.mkdir(parents=True, exist_ok=True)
    
    def write_record(self, record_type: str, record_id: str,
                     payload: Mapping[str, object]) -> None:
        sanitized = sanitize_for_persistence(payload)
        path = self._root / record_type / f"{record_id}.json"
        path.parent.mkdir(parents=True, exist_ok=True)
        tmp = path.with_suffix(".tmp")
        tmp.write_text(json.dumps(sanitized.value, sort_keys=True,
                                  separators=(",", ":")), encoding="utf-8")
        tmp.replace(path)
    
    def read_record(self, record_type: str, record_id: str):
        path = self._root / record_type / f"{record_id}.json"
        if not path.exists():
            return None
        return json.loads(path.read_text(encoding="utf-8"))
    
    def list_records(self, record_type: str):
        dir_path = self._root / record_type
        if not dir_path.exists():
            return ()
        records = []
        for path in sorted(dir_path.glob("*.json")):
            records.append(json.loads(path.read_text(encoding="utf-8")))
        return tuple(records)
    
    def close(self) -> None:
        pass
