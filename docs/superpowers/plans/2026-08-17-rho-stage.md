> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Retrospective Harness Optimization (RHO) stage to AgentEvolve that selects a difficult-and-diverse coreset from historical traces, diagnoses group rollouts, proposes N independent candidate harnesses through CUGA-SDK workspace agents, ranks them by pairwise self-preference, and retains all N in the persistent pool as parents for the existing genetic loop.

**Architecture:** Two cognition stacks over one shared CUGA-SDK invocation layer. Interface A = structured stateless LLM call (comprehension, difficulty/fingerprint). Interface B = CUGA-SDK workspace agent with tools and staged-artifact capture (diagnosis, optimization, preference judging). Embeddings, DPP, and aggregation are deterministic Python. `core/` stays agent-neutral; `adapters/` holds all CUGA specifics.

**Tech Stack:** Python 3.14, `uv`, pytest, numpy, litellm (Interface A), CUGA SDK (Interface B), Ollama embeddings with lexical fallback.

**Spec:** `docs/superpowers/specs/2026-08-17-rho-stage-design.md`

```python
"""Historical (task, trajectory) corpus loading for the RHO coreset stage.

RHO is retrospective: it needs past trajectories to judge difficulty and
fingerprint failure structure. Two trace formats exist in this repository and
only one is usable.

A *stale-format* trace carries eight generic ``stream_event`` entries with
``actor_id=None`` and no ``tool_call`` events. Group diagnosis cannot attribute
anything in such a trace, so accepting it would produce confident diagnoses
about an agent whose behaviour was never recorded. It is rejected with a reason
instead.

When no valid record is found the loader reports a *cold start* rather than
raising: RHO can still run by selecting a coreset without difficulty weighting
and generating its own evidence through group rollouts.
"""
from __future__ import annotations

import json
from dataclasses import dataclass
from hashlib import sha256
from pathlib import Path
from typing import Mapping

TRACE_FILENAME = "causal-trace.json"


class StaleTraceFormat(ValueError):
    """A trace exists but cannot support diagnosis."""


@dataclass(frozen=True, slots=True)
class HistoricalRecord:
    """One past (task, trajectory) pair usable as RHO evidence."""

    task_id: str
    input_text: str
    trace_path: str
    raw_trace: Mapping[str, object]
    final_output: str
    tool_observation_count: int
    harness_version: str
    content_hash: str


@dataclass(frozen=True, slots=True)
class HistoryLoadReport:
    """Loaded records plus every rejection, so nothing is silently dropped."""

    records: tuple[HistoricalRecord, ...] = ()
    rejected: tuple[tuple[str, str], ...] = ()

    @property
    def is_cold_start(self) -> bool:
        return not self.records


def _validate_current_format(payload: Mapping[str, object]) -> None:
    """Raise :class:`StaleTraceFormat` for a trace that cannot be diagnosed."""
    events = payload.get("events")
    if not isinstance(events, list) or not events:
        raise StaleTraceFormat("trace has no events")
    actors = {
        e.get("actor_id") for e in events if isinstance(e, Mapping)
    }
    if actors == {None}:
        raise StaleTraceFormat(
            "every event has actor_id=None: stale trace format, nothing can be "
            "causally attributed"
        )
    kinds = {e.get("kind") for e in events if isinstance(e, Mapping)}
    if kinds == {"stream_event"}:
        raise StaleTraceFormat(
            "every event kind is 'stream_event': stale trace format"
        )


def load_history(root: Path) -> HistoryLoadReport:
    """Load every ``<root>/<run_id>/causal-trace.json`` under ``root``."""
    if not root.exists():
        return HistoryLoadReport()

    records: list[HistoricalRecord] = []
    rejected: list[tuple[str, str]] = []
    for path in sorted(root.rglob(TRACE_FILENAME)):
        try:
            raw = path.read_text(encoding="utf-8")
            payload = json.loads(raw)
        except (OSError, json.JSONDecodeError) as exc:
            rejected.append((str(path), f"unreadable: {exc}"))
            continue
        if not isinstance(payload, dict):
            rejected.append((str(path), "trace root is not an object"))
            continue
        try:
            _validate_current_format(payload)
        except StaleTraceFormat as exc:
            rejected.append((str(path), str(exc)))
            continue
        observations = payload.get("tool_observations")
        records.append(
            HistoricalRecord(
                task_id=str(payload.get("task_id") or path.parent.name),
                input_text=str(payload.get("input_text") or ""),
                trace_path=str(path),
                raw_trace=payload,
                final_output=str(payload.get("final_output") or ""),
                tool_observation_count=(
                    len(observations) if isinstance(observations, list) else 0
                ),
                harness_version=str(payload.get("harness_version") or "unknown"),
                content_hash=f"sha256:{sha256(raw.encode()).hexdigest()}",
            )
        )
    return HistoryLoadReport(records=tuple(records), rejected=tuple(rejected))
```

```python
"""Content-hash keyed JSON cache for RHO preprocessing stages.

Comprehension and difficulty judging are the only RHO stages that re-run over
the same historical corpus every round, so they are the only stages worth
caching. Keys always include the trace content hash, which is why a stale trace
can never produce a false hit.

A ``None`` root disables the cache entirely: a measurement run must be able to
spend nothing on capture, and creating directories for a run that asked for no
cache would violate that.
"""
from __future__ import annotations

import json
from dataclasses import dataclass, field
from hashlib import sha256
from pathlib import Path
from typing import Mapping


@dataclass(slots=True)
class JsonDiskCache:
    """A tiny filename-per-key JSON cache with hit/miss counters."""

    root: Path | None
    hits: int = field(default=0, init=False)
    misses: int = field(default=0, init=False)

    def _path(self, key: str) -> Path | None:
        if self.root is None:
            return None
        digest = sha256(key.encode("utf-8")).hexdigest()
        return self.root / f"{digest}.json"

    def get(self, key: str) -> dict | None:
        path = self._path(key)
        if path is None or not path.exists():
            self.misses += 1
            return None
        try:
            value = json.loads(path.read_text(encoding="utf-8"))
        except (OSError, json.JSONDecodeError):
            # A corrupt entry must not abort a round; recompute instead.
            self.misses += 1
            return None
        if not isinstance(value, dict):
            self.misses += 1
            return None
        self.hits += 1
        return value

    def put(self, key: str, value: Mapping[str, object]) -> None:
        path = self._path(key)
        if path is None:
            return
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(json.dumps(dict(value), default=str), encoding="utf-8")
```

```python
"""Interface A: turn a raw causal trace into a bounded semantic summary.

Why this stage exists at all
----------------------------
Measured on the canonical current-format trace
(``data/traces/0cb88c5a-.../causal-trace.json``, 9,610 bytes, 19 events):
40 UUIDs (1,440 bytes), 20 long hex hashes (1,280 bytes) and 260 JSON keys
(3,121 bytes) account for **60.8%** of the payload before braces and quotes.
Every trace shares the same schema vocabulary, so embedding raw traces makes
cosine similarity saturate near-uniformly high and the DPP diversity term stops
discriminating -- leaving coreset selection driven by difficulty alone.

Mechanical truncation does not fix this: it removes prose and keeps the schema
noise. So a model call converts the trace into prose about *behaviour*, and that
prose is what gets embedded. The previous Gaia RHO reached the same conclusion
(``trajectory_summary.md`` as the "preferred input to embedding").

This is an ordinary structured LLM call (Interface A), not a workspace agent:
the stage needs a bounded abstraction, not filesystem inspection.
"""
from __future__ import annotations

import json

COMPREHENDER_MODEL_ID = "cuga-rho-comprehender"

_SYSTEM_PROMPT = """\
You summarize one agent trajectory for later comparison with other trajectories.

Return ONLY a JSON object with exactly these keys:
  what_was_attempted   one sentence: the goal the agent pursued
  approach_taken       one or two sentences: the strategy actually observed
  where_it_went_wrong  one or two sentences: the first decisive failure, or "" if it succeeded
  tools_used           array of tool names that actually executed (may be empty)
  outcome              one of: correct_answer, wrong_answer, no_committed_answer, error

Hard rules:
- Write prose about BEHAVIOUR. Never copy identifiers.
- Never include UUIDs, hex hashes, event ids, node ids, or JSON key names.
- Never quote the expected answer.
- Judge tool use by observed tool execution, not by what the agent said it would do.
"""


@dataclass(frozen=True, slots=True)
class TrajectorySummary:
    """A bounded, identifier-free semantic summary of one trajectory."""

    task_id: str
    what_was_attempted: str = ""
    approach_taken: str = ""
    where_it_went_wrong: str = ""
    tools_used: tuple[str, ...] = ()
    outcome: str = ""
    observed: bool = False
    error: str = ""

    @property
    def embedding_text(self) -> str:
        """The text that gets embedded for DPP diversity.

        Deliberately excludes ``task_id``: an id is exactly the kind of
        surface token that would make two structurally different failures look
        different for the wrong reason.
        """
        parts = [
            self.what_was_attempted,
            self.approach_taken,
            self.where_it_went_wrong,
            " ".join(self.tools_used),
            self.outcome,
        ]
        return "\n".join(p for p in parts if p)


def _response_text(response: object) -> str:
    """First assistant text from an OpenAI/litellm-shaped response."""
    choices = getattr(response, "choices", None)
    if choices is None and isinstance(response, Mapping):
        choices = response.get("choices")
    if not choices:
        return ""
    first = choices[0]
    message = getattr(first, "message", None)
    if message is None and isinstance(first, Mapping):
        message = first.get("message")
    if message is None:
        return ""
    content = getattr(message, "content", None)
    if content is None and isinstance(message, Mapping):
        content = message.get("content")
    return str(content or "")


def _litellm_completion(**request: object) -> object:
    import litellm

    return litellm.completion(**request)


@dataclass(slots=True)
class RhoComprehender:
    """Interface A trajectory comprehender."""

    completion_fn: Callable[..., object] | None = None
    model: str | None = None
    base_url: str | None = None
    api_key: str | None = None
    #: Omitted unless set. NEVER pass 0.0: the endpoint rejects it outright.
    temperature: float | None = None
    cache: JsonDiskCache = field(default_factory=lambda: JsonDiskCache(None))
    max_events: int = 40

    def _request(self, record: HistoricalRecord) -> dict[str, object]:
        request: dict[str, object] = {"model": self.model or "unset"}
        if self.temperature is not None:
            if self.temperature == 0.0:
                raise ValueError(
                    "temperature=0.0 is rejected by the endpoint; omit it instead"
                )
            request["temperature"] = self.temperature
        if self.base_url:
            request["api_base"] = self.base_url
        if self.api_key:
            request["api_key"] = self.api_key
        request["messages"] = [
            {"role": "system", "content": _SYSTEM_PROMPT},
            {"role": "user", "content": self._user_prompt(record)},
        ]
        return request

    def _user_prompt(self, record: HistoricalRecord) -> str:
        events = record.raw_trace.get("events")
        listed = events[: self.max_events] if isinstance(events, list) else []
        return (
            f"TASK:\n{record.input_text}\n\n"
            f"FINAL OUTPUT:\n{record.final_output}\n\n"
            f"TOOL OBSERVATIONS RECORDED: {record.tool_observation_count}\n\n"
            f"EVENTS (first {len(listed)}):\n{json.dumps(listed, default=str)}"
        )

    def _cache_key(self, record: HistoricalRecord) -> str:
        return (
            f"{COMPREHENDER_MODEL_ID}|{self.model}|{record.task_id}|"
            f"{record.content_hash}"
        )

    def comprehend(self, record: HistoricalRecord) -> TrajectorySummary:
        """Summarize one record, returning an unobserved summary on failure."""
        key = self._cache_key(record)
        cached = self.cache.get(key)
        if cached is not None:
            return TrajectorySummary(
                task_id=record.task_id,
                what_was_attempted=str(cached.get("what_was_attempted", "")),
                approach_taken=str(cached.get("approach_taken", "")),
                where_it_went_wrong=str(cached.get("where_it_went_wrong", "")),
                tools_used=tuple(cached.get("tools_used") or ()),
                outcome=str(cached.get("outcome", "")),
                observed=True,
            )

        invoke = self.completion_fn or _litellm_completion
        try:
            response = invoke(**self._request(record))
        except Exception as exc:  # noqa: BLE001 - a failure is data
            return TrajectorySummary(
                task_id=record.task_id,
                error=f"{type(exc).__name__}: {exc}",
            )

        text = _response_text(response)
        try:
            payload = json.loads(text)
            if not isinstance(payload, dict):
                raise ValueError("response JSON is not an object")
        except (json.JSONDecodeError, ValueError) as exc:
            return TrajectorySummary(
                task_id=record.task_id,
                error=f"unparseable comprehension response: {exc}",
            )

        tools = payload.get("tools_used")
        summary = TrajectorySummary(
            task_id=record.task_id,
            what_was_attempted=str(payload.get("what_was_attempted", "")),
            approach_taken=str(payload.get("approach_taken", "")),
            where_it_went_wrong=str(payload.get("where_it_went_wrong", "")),
            tools_used=tuple(str(t) for t in tools) if isinstance(tools, list) else (),
            outcome=str(payload.get("outcome", "")),
            observed=True,
        )
        self.cache.put(
            key,
            {
                "what_was_attempted": summary.what_was_attempted,
                "approach_taken": summary.approach_taken,
                "where_it_went_wrong": summary.where_it_went_wrong,
                "tools_used": list(summary.tools_used),
                "outcome": summary.outcome,
            },
        )
        return summary
```

```python
"""Interface A: historical-task difficulty and abstract fingerprint.

Matches the paper's own split: difficulty scoring and fingerprint generation are
a conventional structured LLM request, not an agentic invocation, because the
stage needs only a bounded classification and abstraction result
(``RHO_agents_context.md:194-236``).

The judge consumes the *comprehended summary*, not the raw trace. That keeps the
prompt bounded and keeps raw ``final_output`` -- which may contain the expected
answer -- out of the request.

The fingerprint must describe structural form and failure shape without
task-specific identifiers, so that cosine similarity between fingerprints
reflects failure structure rather than shared project vocabulary.
"""
from __future__ import annotations

import json
from dataclasses import dataclass, field
from typing import Callable

from agent_evolve.adapters.cuga_rho_comprehender import (
    TrajectorySummary,
    _litellm_completion,
    _response_text,
)
from agent_evolve.core.rho.cache import JsonDiskCache
from agent_evolve.core.rho.history import HistoricalRecord

JUDGE_MODEL_ID = "cuga-rho-difficulty-judge"

MIN_DIFFICULTY = 0.0
MAX_DIFFICULTY = 10.0

_SYSTEM_PROMPT = """\
You rate how difficult a task was for an agent, and you abstract its failure shape.

Return ONLY a JSON object with exactly these keys:
  difficulty            number in [0, 10]; 0 trivial, 10 defeated the agent
  abstract_fingerprint  2-3 sentences describing the STRUCTURAL form of the task
                        and its failure mode

Hard rules for abstract_fingerprint:
- Describe structure, not content. "A multi-step retrieval task where the agent
  plans in prose but never executes a retrieval step" is correct.
- Never name files, repositories, frameworks, functions, people, or places.
- Never include the answer or any value from the task.
- Two tasks with the same failure structure must produce similar fingerprints.
"""


@dataclass(frozen=True, slots=True)
class DifficultyVerdict:
    """One judged historical record."""

    task_id: str
    difficulty: float = 0.0
    abstract_fingerprint: str = ""
    observed: bool = False
    error: str = ""


@dataclass(slots=True)
class RhoDifficultyJudge:
    """Interface A difficulty and fingerprint judge."""

    completion_fn: Callable[..., object] | None = None
    model: str | None = None
    base_url: str | None = None
    api_key: str | None = None
    #: Omitted unless set. NEVER pass 0.0: the endpoint rejects it outright.
    temperature: float | None = None
    cache: JsonDiskCache = field(default_factory=lambda: JsonDiskCache(None))

    def _request(self, summary: TrajectorySummary) -> dict[str, object]:
        request: dict[str, object] = {"model": self.model or "unset"}
        if self.temperature is not None:
            if self.temperature == 0.0:
                raise ValueError(
                    "temperature=0.0 is rejected by the endpoint; omit it instead"
                )
            request["temperature"] = self.temperature
        if self.base_url:
            request["api_base"] = self.base_url
        if self.api_key:
            request["api_key"] = self.api_key
        request["messages"] = [
            {"role": "system", "content": _SYSTEM_PROMPT},
            {"role": "user", "content": f"TRAJECTORY SUMMARY:\n{summary.embedding_text}"},
        ]
        return request

    def _cache_key(self, record: HistoricalRecord) -> str:
        return f"{JUDGE_MODEL_ID}|{self.model}|{record.task_id}|{record.content_hash}"

    def judge(
        self, record: HistoricalRecord, summary: TrajectorySummary
    ) -> DifficultyVerdict:
        """Judge one record, returning an unobserved verdict on any problem."""
        key = self._cache_key(record)
        cached = self.cache.get(key)
        if cached is not None:
            return DifficultyVerdict(
                task_id=record.task_id,
                difficulty=float(cached.get("difficulty", 0.0)),
                abstract_fingerprint=str(cached.get("abstract_fingerprint", "")),
                observed=True,
            )

        invoke = self.completion_fn or _litellm_completion
        try:
            response = invoke(**self._request(summary))
        except Exception as exc:  # noqa: BLE001 - a failure is data
            return DifficultyVerdict(
                task_id=record.task_id, error=f"{type(exc).__name__}: {exc}"
            )

        try:
            payload = json.loads(_response_text(response))
            if not isinstance(payload, dict):
                raise ValueError("response JSON is not an object")
        except (json.JSONDecodeError, ValueError) as exc:
            return DifficultyVerdict(
                task_id=record.task_id,
                error=f"unparseable judge response: {exc}",
            )

        raw = payload.get("difficulty")
        try:
            difficulty = float(raw)  # type: ignore[arg-type]
        except (TypeError, ValueError):
            return DifficultyVerdict(
                task_id=record.task_id,
                error=f"difficulty is not a number: {raw!r}",
            )
        if not MIN_DIFFICULTY <= difficulty <= MAX_DIFFICULTY:
            return DifficultyVerdict(
                task_id=record.task_id,
                error=(
                    f"difficulty {difficulty} outside "
                    f"[{MIN_DIFFICULTY}, {MAX_DIFFICULTY}]"
                ),
            )

        fingerprint = str(payload.get("abstract_fingerprint") or "").strip()
        if not fingerprint:
            return DifficultyVerdict(
                task_id=record.task_id,
                error="abstract_fingerprint is missing or empty",
            )

        self.cache.put(
            key,
            {"difficulty": difficulty, "abstract_fingerprint": fingerprint},
        )
        return DifficultyVerdict(
            task_id=record.task_id,
            difficulty=difficulty,
            abstract_fingerprint=fingerprint,
            observed=True,
        )
```

```python
Quality is judge-assigned difficulty; diversity is cosine over fingerprint
embeddings. This is NOT the genetic issue-selection quality function (which is
cross-candidate variance) -- only the DPP primitives are shared.
"""

from agent_evolve.core.clustering import EmbeddingProviderUnavailable
from agent_evolve.core.rho.coreset import (
    CoresetCandidate,
    select_coreset,
)


class _FixedEmbedder:
    """Maps a marker token to an orthogonal-ish unit vector."""

    dim = 4

    def embed(self, text: str) -> tuple[float, ...]:
        if "ALPHA" in text:
            return (1.0, 0.0, 0.0, 0.0)
        if "BETA" in text:
            return (0.0, 1.0, 0.0, 0.0)
        return (0.0, 0.0, 1.0, 0.0)


class _BrokenEmbedder:
    dim = 4

    def embed(self, text: str) -> tuple[float, ...]:
        raise EmbeddingProviderUnavailable("ollama down")


def _c(task_id: str, difficulty: float, marker: str) -> CoresetCandidate:
    return CoresetCandidate(
        task_id=task_id,
        difficulty=difficulty,
        fingerprint=f"{marker} shaped failure",
        embedding_text=f"{marker} shaped failure",
    )


def test_prefers_higher_difficulty() -> None:
    candidates = (
        _c("easy", 1.0, "ALPHA"),
        _c("hard", 9.0, "ALPHA"),
    )

    report = select_coreset(candidates, 1, embedder=_FixedEmbedder())

    assert report.selected_ids == ("hard",)
```

```python
"""RHO coreset selection: difficulty-weighted diversity over historical records.

RHO does not diagnose every historical trajectory. It selects a small coreset
that is simultaneously:

1. difficult, because difficult trajectories expose harness weakness, and
2. diverse, so the optimizer does not learn from one narrow failure mode.

The DPP primitives are shared with genetic issue selection
(``build_kernel``/``greedy_map`` in :mod:`agent_evolve.core.issues`); only the
*quality vector* differs. Here quality is judge-assigned difficulty. In genetic
issue selection quality is cross-candidate score variance. Those two must not be
unified: they answer different questions.

``embedding_text`` comes from the comprehended trajectory summary, never the raw
trace. A raw trace is ~60% identifiers and schema keys, which saturates cosine
similarity and neutralizes the diversity term.
"""
from __future__ import annotations

import math
import random
from dataclasses import dataclass
from typing import Sequence

from agent_evolve.core.clustering import (
    EmbeddingProviderUnavailable,
    MechanismEmbedder,
)
from agent_evolve.core.issues import build_kernel, greedy_map

DEFAULT_THETA = 0.7
DEFAULT_SCORE_FLOOR = 0.1
DEFAULT_MIN_GAIN = 1e-12
DEFAULT_JITTER = 1e-9
MAX_DIFFICULTY = 10.0

SELECTORS = ("dpp", "difficulty_rank", "random")


@dataclass(frozen=True, slots=True)
class CoresetCandidate:
    """One historical record eligible for coreset selection."""

    task_id: str
    difficulty: float
    fingerprint: str
    embedding_text: str


@dataclass(frozen=True, slots=True)
class CoresetReport:
    """The selected coreset plus how it was selected."""

    selected_ids: tuple[str, ...] = ()
    selection_method: str = "dpp"
    fallback_reason: str = ""


def _cosine(a: Sequence[float], b: Sequence[float]) -> float:
    dot = sum(x * y for x, y in zip(a, b))
    na = math.sqrt(sum(x * x for x in a))
    nb = math.sqrt(sum(y * y for y in b))
    if na == 0.0 or nb == 0.0:
        return 0.0
    return dot / (na * nb)


def _quality(difficulty: float, theta: float, score_floor: float) -> float:
    """Floor, normalize to [0,1], then exponentiate by the theta exponent.

    Mirrors ``issues.quality`` so RHO and genetic selection sharpen quality the
    same way; only the input signal differs.
    """
    normalized = max(difficulty / MAX_DIFFICULTY, score_floor)
    alpha = theta / (2 * max(1 - theta, 1e-6)) if theta < 1.0 else 1.0
    return normalized**alpha


def select_coreset(
    candidates: Sequence[CoresetCandidate],
    k: int,
    *,
    selector: str = "dpp",
    theta: float = DEFAULT_THETA,
    score_floor: float = DEFAULT_SCORE_FLOOR,
    min_gain: float = DEFAULT_MIN_GAIN,
    seed: int = 0,
    embedder: MechanismEmbedder | None = None,
) -> CoresetReport:
    """Select up to ``k`` task ids from ``candidates``."""
    if selector not in SELECTORS:
        raise ValueError(f"unknown selector: {selector!r}; expected one of {SELECTORS}")
    if k < 1:
        raise ValueError("k must be >= 1")
    if not candidates:
        return CoresetReport(selection_method=selector)

    if selector == "difficulty_rank":
        ordered = sorted(
            candidates, key=lambda c: (-c.difficulty, c.task_id)
        )
        return CoresetReport(
            selected_ids=tuple(c.task_id for c in ordered[:k]),
            selection_method="difficulty_rank",
        )

    if selector == "random":
        rng = random.Random(seed)
        pool = sorted(candidates, key=lambda c: c.task_id)
        picked = rng.sample(pool, min(k, len(pool)))
        return CoresetReport(
            selected_ids=tuple(c.task_id for c in picked),
            selection_method="random",
        )

    # --- dpp ---------------------------------------------------------- #
    qualities = [_quality(c.difficulty, theta, score_floor) for c in candidates]
    ids = tuple(c.task_id for c in candidates)

    fallback_reason = ""
    vectors: list[tuple[float, ...]] | None = None
    if embedder is not None:
        try:
            vectors = [embedder.embed(c.embedding_text) for c in candidates]
        except EmbeddingProviderUnavailable as exc:
            # Degrade to quality-only ordering, but record why. Silent
            # substitution would make a degraded run indistinguishable from a
            # healthy one.
            fallback_reason = f"embedding provider unavailable: {exc}"
            vectors = None

    if vectors is None:
        ordered = sorted(candidates, key=lambda c: (-c.difficulty, c.task_id))
        return CoresetReport(
            selected_ids=tuple(c.task_id for c in ordered[:k]),
            selection_method="dpp_quality_only",
            fallback_reason=fallback_reason,
        )

    def similarity(i: int, j: int) -> float:
        return _cosine(vectors[i], vectors[j])

    kernel = build_kernel(qualities, similarity, DEFAULT_JITTER)
    chosen = greedy_map(kernel, ids, k, min_gain)
    return CoresetReport(
        selected_ids=tuple(ids[i] for i in chosen),
        selection_method="dpp",
        fallback_reason=fallback_reason,
    )
```

```python
"""Shared Interface B mechanism: run a CUGA workspace agent over tool callables.

RHO's group diagnoser, candidate optimizer, and preference judge are all
workspace agents in the published implementation. They differ in prompt and tool
set, not in mechanism, so the mechanism lives here once.

Two layers, mirroring ``cuga_editor_tools``:

* the caller supplies plain callables with NO CUGA dependency, so authorization
  and capture rules stay unit-testable offline, and
* this module wraps them with ``tracked_tool`` + ``tool`` and defers the SDK
  import into the function body.

Tool calls are recorded by wrapping the callables, because the agent's prose is
not evidence that a tool ran. That is the same tool-execution-over-narration
principle the editor already relies on.

``cuga_editor`` is deliberately NOT refactored onto this helper: the genetic path
works, and changing it under deadline would risk a measured result for a
cosmetic gain.
"""
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import Callable, Mapping


@dataclass(frozen=True, slots=True)
class WorkspaceAgentRun:
    """The outcome of one workspace-agent invocation."""

    answer: str = ""
    tools_called: tuple[str, ...] = ()
    sdk_tool_calls: tuple = ()
    error: str = ""

    @property
    def ok(self) -> bool:
        return not self.error


def _recording_wrapper(
    callables: Mapping[str, Callable[..., str]]
) -> tuple[dict[str, Callable[..., str]], list[str]]:
    """Wrap each callable so an actual invocation is recorded by name."""
    names: list[str] = []

    def wrap(name: str, fn: Callable[..., str]) -> Callable[..., str]:
        def recorded(*args: object, **kwargs: object) -> str:
            names.append(name)
            return fn(*args, **kwargs)

        recorded.__name__ = name
        return recorded

    return {name: wrap(name, fn) for name, fn in callables.items()}, names


def _build_tracked_tools(
    callables: Mapping[str, Callable[..., str]], app_names: Mapping[str, str]
) -> list:
    """Wrap recorded callables as tracked LangChain tools. SDK import is local."""
    from langchain_core.tools import tool

    from cuga import tracked_tool

    built = []
    for name, fn in callables.items():
        fn.__name__ = name
        wrapped = tracked_tool(app_name=app_names.get(name, "rho"))(fn)
        built.append(tool(wrapped))
    return built


def _run_real_agent(
    callables: Mapping[str, Callable[..., str]],
    prompt: str,
    app_names: Mapping[str, str],
    skills_dir: Path | None,
) -> tuple[str, tuple]:
    """Construct and run a real CugaAgent. SDK import stays local."""
    import asyncio

    from cuga import CugaAgent

    kwargs: dict[str, object] = {}
    if skills_dir is not None:
        kwargs["skills_dir"] = str(skills_dir)
    agent = CugaAgent(
        tools=_build_tracked_tools(callables, app_names), **kwargs
    )

    async def run() -> tuple[str, tuple]:
        await agent.initialize()
        result = await agent.invoke(prompt, track_tool_calls=True)
        return str(result), tuple(getattr(result, "tool_calls", ()) or ())

    return asyncio.run(run())


def run_workspace_agent(
    callables: Mapping[str, Callable[..., str]],
    prompt: str,
    *,
    app_names: Mapping[str, str],
    skills_dir: Path | None = None,
    agent_factory: Callable[[dict, str], str] | None = None,
) -> WorkspaceAgentRun:
    """Run one workspace agent, returning failure as data rather than raising.

    A raised exception here would discard the whole round's evidence for one bad
    invocation, so every failure is classified and returned.
    """
    recorded, names = _recording_wrapper(callables)
    try:
        if agent_factory is not None:
            answer = agent_factory(recorded, prompt)
            sdk_calls: tuple = ()
        else:
            answer, sdk_calls = _run_real_agent(
                recorded, prompt, app_names, skills_dir
            )
    except Exception as exc:  # noqa: BLE001 - a failure is data
        return WorkspaceAgentRun(
            tools_called=tuple(names), error=f"{type(exc).__name__}: {exc}"
        )
    return WorkspaceAgentRun(
        answer=str(answer),
        tools_called=tuple(names),
        sdk_tool_calls=sdk_calls,
    )
```

```python
"""Interface B: diagnose one task's G rollouts in a single workspace-agent call.

Per ``RHO_agents_context.md:59-94``, self-validation and self-consistency are two
signals extracted *within one diagnosis invocation*, not separate calls. With
k=10 that is 10 invocations, not 20.

The agent gets tools to inspect each rollout selectively rather than one enormous
serialized prompt. That is the reason the paper uses an agent here, and it keeps
the prompt bounded when G=3 trajectories of 19-56 events each would otherwise be
concatenated.

The prompt states the real SDK graph shape and must distinguish "narrated without
emitting an executable code block, so sandbox was never reached" from a genuine
tool failure. Ground truth for tool execution is recorded tool observations,
never model prose.
"""
from __future__ import annotations

import json
from dataclasses import dataclass, field
from typing import Callable, Mapping, Sequence

from agent_evolve.adapters.cuga_workspace_agent import run_workspace_agent
from agent_evolve.core.contracts import ExecutionTrace

DIAGNOSER_MODEL_ID = "cuga-rho-group-diagnoser"

#: The four surfaces CugaAdapter can actually deliver.
VALID_SURFACES = frozenset({"instructions", "skills", "policies", "memory"})

APP_NAMES: dict[str, str] = {
    "get_task": "evidence",
    "list_rollouts": "evidence",
    "read_rollout_events": "evidence",
    "submit_diagnosis": "submit",
}

_PROMPT = """\
You are diagnosing why an agent harness underperformed on ONE task.

The agent under study runs this graph:
  CugaLiteSubgraph -> prepare -> call_model <-> sandbox -> SDKCallback -> FinalAnswerAgent
A tool runs ONLY when call_model emits an executable fenced Python block that
reaches `sandbox`. Only the FIRST fenced block per turn executes.

You have {count} independent rollouts of the SAME task with the SAME harness.

Use the tools to inspect them:
  get_task()                      the task input
  list_rollouts()                 ids, final outputs, tool-observation counts
  read_rollout_events(rollout_id) that rollout's events

Then analyse:
1. Per rollout, did the agent verify its own answer before committing?
   (self-validation)
2. Across rollouts, where do they meaningfully DISAGREE in interpretation, plan,
   actions, or final answer? Distinguish harmless variation from consequential
   divergence. (self-consistency)
3. What failure mode RECURS across rollouts?
4. How severe is the harness weakness, in [0, 1]?
5. What general harness improvement would address it?

Critical distinction: if a rollout produced NO tool observations and the text is
narration about what it would do, the failure is "never emitted executable code",
NOT "the tool failed". Never claim a tool failed unless a tool observation shows
it failing.

Finish by calling submit_diagnosis exactly once with:
  recurring_failure_mode     one or two sentences
  disagreements              list of consequential disagreements (may be empty)
  self_validation_observed   true only if some rollout really verified its answer
  severity                   number in [0, 1]
  improvement_direction      general, not task-specific
  candidate_surfaces         any of: instructions, skills, policies, memory
"""


@dataclass(frozen=True, slots=True)
class GroupDiagnosis:
    """One task's diagnosis over its whole rollout group."""

    task_id: str
    recurring_failure_mode: str = ""
    disagreements: tuple[str, ...] = ()
    self_validation_observed: bool = False
    severity: float = 0.0
    improvement_direction: str = ""
    candidate_surfaces: tuple[str, ...] = ()
    rollouts_seen: int = 0
    observed: bool = False
    error: str = ""


def _build_callables(
    task_input: str, traces: Sequence[ExecutionTrace], plan: dict
) -> dict[str, Callable[..., str]]:
    def get_task() -> str:
        """Return the task input text."""
        return json.dumps({"input": task_input})

    def list_rollouts() -> str:
        """List every rollout with its final output and tool-observation count."""
        return json.dumps(
            {
                "rollouts": [
                    {
                        "rollout_id": t.trace_id,
                        "final_output": t.final_output,
                        "status": t.status,
                        "event_count": len(t.events),
                    }
                    for t in traces
                ]
            }
        )

    def read_rollout_events(rollout_id: str) -> str:
        """Return one rollout's events."""
        for t in traces:
            if t.trace_id == rollout_id:
                return json.dumps(
                    {
                        "events": [
                            {

                            }
                            for e in t.events

                    },
                    default=str,
                )
        return json.dumps({"status": "error", "message": f"unknown {rollout_id!r}"})

    def submit_diagnosis(
        recurring_failure_mode: str,
        disagreements: list | None = None,
        self_validation_observed: bool = False,
        severity: float = 0.0,
        improvement_direction: str = "",

            "recurring_failure_mode": recurring_failure_mode,
            "disagreements": disagreements or [],
            "self_validation_observed": self_validation_observed,
            "severity": severity,
            "improvement_direction": improvement_direction,
            "candidate_surfaces": candidate_surfaces or [],

        "get_task": get_task,
        "list_rollouts": list_rollouts,

def _validate(payload: Mapping[str, object]) -> str:
    """Return an error string, or "" when the payload is acceptable."""
    raw = payload.get("severity")
    try:
        severity = float(raw)  # type: ignore[arg-type]
    except (TypeError, ValueError):
        return f"severity is not a number: {raw!r}"
    if not 0.0 <= severity <= 1.0:
        return f"severity {severity} outside [0, 1]"
    surfaces = payload.get("candidate_surfaces") or []
    if not isinstance(surfaces, list):
        return "candidate_surfaces must be a list"
    for surface in surfaces:
        if str(surface) not in VALID_SURFACES:
            return (
                f"unknown surface {surface!r}; expected one of "
                f"{sorted(VALID_SURFACES)}"
            )
    if not str(payload.get("recurring_failure_mode") or "").strip():
        return "recurring_failure_mode is empty"
    return ""


@dataclass(slots=True)
class RhoGroupDiagnoser:
    """Interface B group diagnoser: one invocation per task."""

    agent_factory: Callable[[dict, str], str] | None = None
    app_names: Mapping[str, str] = field(default_factory=lambda: dict(APP_NAMES))

    def diagnose(
        self,
        task_id: str,
        task_input: str,
        traces: Sequence[ExecutionTrace],
    ) -> GroupDiagnosis:
        """Diagnose one task's rollout group, returning failure as data."""
        if not traces:
            return GroupDiagnosis(
                task_id=task_id, error="no rollouts to diagnose"
            )

        plan: dict = {}
        callables = _build_callables(task_input, traces, plan)
        run = run_workspace_agent(
            callables,
            _PROMPT.format(count=len(traces)),
            app_names=self.app_names,
            agent_factory=self.agent_factory,
        )
        if not run.ok:
            return GroupDiagnosis(
                task_id=task_id, rollouts_seen=len(traces), error=run.error
            )

        payload = plan.get("payload")
        if payload is None:
            # Fall back to parsing the answer, so a model that returned a JSON
            # object without calling the tool is still usable.
            try:
                parsed = json.loads(run.answer)
                if not isinstance(parsed, dict):
                    raise ValueError("answer JSON is not an object")
                payload = parsed
            except (json.JSONDecodeError, ValueError) as exc:
                return GroupDiagnosis(
                    task_id=task_id,
                    rollouts_seen=len(traces),
                    error=f"no diagnosis submitted and answer unparseable: {exc}",
                )

        error = _validate(payload)
        if error:
            return GroupDiagnosis(
                task_id=task_id, rollouts_seen=len(traces), error=error
            )

        disagreements = payload.get("disagreements") or []
        surfaces = payload.get("candidate_surfaces") or []
        return GroupDiagnosis(
            task_id=task_id,
            recurring_failure_mode=str(payload["recurring_failure_mode"]),
            disagreements=tuple(str(d) for d in disagreements),
            self_validation_observed=bool(payload.get("self_validation_observed")),
            severity=float(payload["severity"]),
            improvement_direction=str(payload.get("improvement_direction") or ""),
            candidate_surfaces=tuple(str(s) for s in surfaces),
            rollouts_seen=len(traces),
            observed=True,
        )
```

```python
Per ``RHO_agents_context.md:125-163`` each candidate is its own agent invocation
with its own workspace:

    candidate_0 = optimizer.run(fresh_workspace_0)
    candidate_1 = optimizer.run(fresh_workspace_1)
    candidate_2 = optimizer.run(fresh_workspace_2)

Diversity therefore comes from independent multi-step tool trajectories, not from
token sampling. Two agents that read different artifacts in a different order and
stage different edits differ for a substantive reason. Sampling temperature is an
ablation knob, not the mechanism, and is omitted by default.

A candidate is captured from **staged artifacts**, never parsed from the agent's
final text: the paper's "captured from the filesystem, rather than parsed from the
optimizer's final textual answer". Unfinalized staging is discarded.

A no-op is not a candidate. Byte-identical-to-base and duplicate artifact sets are
discarded before evaluation so the pairwise judge never compares a harness with
itself.
"""

You are improving a persistent agent harness from accumulated failure evidence.

The agent under study runs:
  CugaLiteSubgraph -> prepare -> call_model <-> sandbox -> SDKCallback -> FinalAnswerAgent
A tool runs ONLY when call_model emits an executable fenced Python block. Only the

  instructions      ALWAYS in context. Use for unconditional turn-level behaviour.

@dataclass(frozen=True, slots=True)
class ProposalReport:
    """Every proposal outcome, so a collapse to one candidate is visible."""

    candidates: tuple[ProposedCandidate, ...] = ()

def _fingerprint(artifacts: Mapping[str, str]) -> str:
    payload = json.dumps(dict(sorted(artifacts.items())), sort_keys=True)
    return sha256(payload.encode()).hexdigest()


def _render_diagnoses(diagnoses: Sequence[GroupDiagnosis]) -> str:
    lines = []
    for d in diagnoses:
        lines.append(
            f"- task {d.task_id} (severity {d.severity:.2f}, "
            f"{d.rollouts_seen} rollouts): {d.recurring_failure_mode} "
            f"| direction: {d.improvement_direction} "
            f"| suggested surfaces: {', '.join(d.candidate_surfaces) or 'none'}"
        )
    return "\n".join(lines) if lines else "- (no observed diagnoses)"


def _build_callables(
    base_artifacts: Mapping[str, str],
    diagnoses: Sequence[GroupDiagnosis],
    staging: EditStagingArea,
    plan: dict,
) -> dict[str, Callable[..., str]]:
    by_id = {d.task_id: d for d in diagnoses}

    def list_diagnoses() -> str:
        """List every task diagnosis, most severe first."""
        return json.dumps(
            {
                "diagnoses": [
                    {
                        "task_id": d.task_id,
                        "severity": d.severity,
                        "recurring_failure_mode": d.recurring_failure_mode,
                        "candidate_surfaces": list(d.candidate_surfaces),
                    }
                    for d in diagnoses
                ]
            }
        )

    def read_diagnosis(task_id: str) -> str:
        """Return one task's diagnosis in full."""
        d = by_id.get(task_id)
        if d is None:
            return json.dumps({"status": "error", "message": f"unknown {task_id!r}"})
        return json.dumps(
            {
                "task_id": d.task_id,
                "recurring_failure_mode": d.recurring_failure_mode,
                "disagreements": list(d.disagreements),
                "self_validation_observed": d.self_validation_observed,
                "severity": d.severity,
                "improvement_direction": d.improvement_direction,
                "candidate_surfaces": list(d.candidate_surfaces),
            }
        )

    def list_artifacts() -> str:
        """List the harness artifacts you may read and edit."""
        return json.dumps({"artifacts": sorted(base_artifacts)})

    def read_artifact(artifact_id: str) -> str:
        """Return one artifact's current content."""
        if artifact_id not in base_artifacts:
            return json.dumps(
                {"status": "error", "message": f"unknown artifact {artifact_id!r}"}
            )
        return json.dumps({"artifact_id": artifact_id, "content": base_artifacts[artifact_id]})

    def stage_replace(artifact_id: str, content: str) -> str:
        """Stage a rewrite of an existing artifact."""
        outcome = staging.stage_replace(artifact_id, content)
        return json.dumps({"accepted": outcome.accepted, "reason": outcome.reason})

    def stage_create(artifact_id: str, content: str) -> str:
        """Stage a brand-new artifact."""
        outcome = staging.stage_create(artifact_id, content)
        return json.dumps({"accepted": outcome.accepted, "reason": outcome.reason})

    def list_staged() -> str:
        """List what you have staged so far."""
        return json.dumps(
            {"staged": [e.artifact_id for e in staging.edits()]}
        )

    def unstage(artifact_id: str) -> str:
        """Drop a staged edit."""
        staging._replaced.pop(artifact_id, None)
        staging._created.pop(artifact_id, None)
        return json.dumps({"status": "ok"})

    def submit_candidate(rationale: str) -> str:
        """Finalize this candidate. Call exactly once."""
        plan["rationale"] = rationale
        return json.dumps({"status": "ok"})

    return {
        "list_diagnoses": list_diagnoses,
        "read_diagnosis": read_diagnosis,
        "list_artifacts": list_artifacts,
        "read_artifact": read_artifact,
        "stage_replace": stage_replace,
        "stage_create": stage_create,
        "list_staged": list_staged,
        "unstage": unstage,
        "submit_candidate": submit_candidate,
    }


@dataclass(slots=True)
class RhoOptimizer:
    """Interface B optimizer: N independent candidate proposals."""

    agent_factory: Callable[[dict, str], str] | None = None
    app_names: Mapping[str, str] = field(default_factory=lambda: dict(APP_NAMES))
    #: Ablation only. NEVER 0.0 -- the endpoint rejects it.
    temperature: float | None = None

    def propose(
        self,
        base_artifacts: Mapping[str, str],
        diagnoses: Sequence[GroupDiagnosis],
        n: int,
    ) -> ProposalReport:
        """Run N independent proposals and return every outcome."""
        if self.temperature is not None and self.temperature == 0.0:
            raise ValueError(
                "temperature=0.0 is rejected by the endpoint; omit it instead"
            )
        if n < 1:
            raise ValueError("n must be >= 1")

        observed = [d for d in diagnoses if d.observed]
        ordered = sorted(observed, key=lambda d: (-d.severity, d.task_id))
        prompt = _PROMPT.format(
            diagnoses=_render_diagnoses(ordered), prefix=CREATABLE_PREFIX
        )

        base_fingerprint = _fingerprint(base_artifacts)
        seen: set[str] = {base_fingerprint}
        survivors: list[ProposedCandidate] = []
        discarded: list[tuple[int, str]] = []

        for index in range(n):
            staging = EditStagingArea(
                write_set=tuple(sorted(base_artifacts)),
                creatable_prefix=CREATABLE_PREFIX,
            )
            plan: dict = {}
            callables = _build_callables(base_artifacts, ordered, staging, plan)
            run = run_workspace_agent(
                callables,
                prompt,
                app_names=self.app_names,
                agent_factory=self.agent_factory,
            )
            if not run.ok:
                discarded.append((index, run.error))
                continue
            if "rationale" not in plan:
                discarded.append(
                    (index, "never called submit_candidate; work discarded")
                )
                continue

            edits = staging.edits()
            if not edits:
                discarded.append((index, "staged no edits; a no-op is not a candidate"))
                continue

            artifacts = dict(base_artifacts)
            for edit in edits:
                artifacts[edit.artifact_id] = str(edit.payload.get("content", ""))

            fingerprint = _fingerprint(artifacts)
            if fingerprint == base_fingerprint:
                discarded.append(
                    (index, "artifact set is identical to the base; no-op discarded")
                )
                continue
            if fingerprint in seen:
                discarded.append((index, "duplicate of an earlier candidate"))
                continue

            seen.add(fingerprint)
            survivors.append(
                ProposedCandidate(
                    candidate_index=index,
                    artifacts=artifacts,
                    rationale=str(plan["rationale"]),
                    tools_called=run.tools_called,
                    observed=True,
                )
            )

        return ProposalReport(
            candidates=tuple(survivors),
            requested=n,
            discarded=tuple(discarded),
        )
```

```python
"""Interface B: signed pairwise preference between baseline and candidate.

Shared by the RHO stage and the genetic stage. Implemented as a workspace agent,
matching the published implementation (``RHO_agents_context.md:165-192``): the
judge gets an evaluation workspace holding the task, both harnesses' trajectories,
and tools to inspect them before committing to a verdict.

The score is SIGNED and describes the transition
``baseline harness -> candidate harness``: positive favours the candidate,
negative favours the baseline, zero is a tie.

Ground truth
------------
Task metadata including ground truth is supplied when available. This is a
deliberate, recorded deviation from the ``AGENTS.md`` no-labels rule (see spec
section 7): containment is by prompting rather than by a hard firewall.

Not every split has ground truth. ``gaia_l1_test.json`` and
``gaia_l1_test_tiny10.json`` carry a single shared regex of ``(?i)\\?`` for every
task, which matches any question mark and so passes vacuously. That is a
placeholder, not an answer, and is filtered here so the judge is never told a
stub is ground truth.

An unparseable or out-of-range verdict is recorded as UNAVAILABLE and contributes
nothing to a candidate's average. Defaulting it to a tie would silently pull every
candidate toward its baseline.
"""
from __future__ import annotations

import json
from dataclasses import dataclass, field
from typing import Callable, Mapping

from agent_evolve.adapters.cuga_workspace_agent import run_workspace_agent
from agent_evolve.core.contracts import EvolutionTask, ExecutionTrace

JUDGE_MODEL_ID = "cuga-preference-judge"

#: Regexes that are stubs rather than ground truth. ``(?i)\?`` matches any
#: question mark, so every answer "passes" -- treating it as GT would make the
#: judge confidently wrong.
PLACEHOLDER_REGEXES = frozenset({r"(?i)\?", r"\?", r"(?i)\\?"})

MIN_SCORE = -1.0
MAX_SCORE = 1.0

APP_NAMES: dict[str, str] = {
    "get_task": "evidence",
    "read_baseline": "evidence",
    "read_candidate": "evidence",
    "submit_preference": "submit",
}

_PROMPT = """\
You are comparing TWO trajectories for the SAME task, produced by two harnesses.

  baseline harness  -> the current reference
  candidate harness -> a proposed replacement

Decide which harness produced the better trajectory, and by how much.

Tools:
  get_task()          the task, and the expected answer IF it is available
  read_baseline()     the baseline trajectory
  read_candidate()    the candidate trajectory
  submit_preference(score, rationale)   finalize ONCE

Scoring, signed, in [-1, 1]:
  > 0   the candidate is better; 1.0 is decisively better
  = 0   genuinely indistinguishable
  < 0   the candidate is worse; -1.0 is decisively worse

{gt_guidance}

Judge the TRAJECTORY, not just the final string. A correct answer reached by
luck after no verification is weaker than the same answer reached by checking.
An answer that was never committed at all is not a better answer.

You MUST call submit_preference, or your comparison is discarded.
"""

_GT_PRESENT = """\
The expected answer IS available from get_task(). Use it: a trajectory that
reaches the correct answer is better than one that does not.
"""

_GT_ABSENT = """\
NO expected answer is available for this task. Do not guess one. Compare on
process quality only: which trajectory reasoned more soundly, executed the tools
it needed, verified its own work, and committed a definite answer.
"""


@dataclass(frozen=True, slots=True)
class PreferenceVerdict:
    """One signed pairwise comparison."""

    task_id: str
    score: float = 0.0
    winner: str = "unavailable"
    rationale: str = ""
    gt_available: bool = False
    available: bool = False
    error: str = ""


def _expected_answer(task: EvolutionTask) -> str:
    """Extract usable ground truth, filtering known placeholders."""
    for key in ("expected_regex", "expected_substring", "expected_answer"):
        raw = task.expected_contract.get(key)
        if raw is None:
            continue
        value = str(raw).strip()
        if not value or value in PLACEHOLDER_REGEXES:
            continue
        return value
    return ""


def _build_callables(
    task: EvolutionTask,
    baseline: ExecutionTrace,
    candidate: ExecutionTrace,
    expected: str,
    plan: dict,
) -> dict[str, Callable[..., str]]:
    def get_task() -> str:
        """Return the task input and the expected answer if available."""
        payload: dict[str, object] = {"input": task.input_text}
        if expected:
            payload["expected_answer"] = expected
        return json.dumps(payload)

    def _render(trace: ExecutionTrace) -> str:
        return json.dumps(
            {
                "final_output": trace.final_output,
                "status": trace.status,
                "events": [
                    {
                        "kind": e.kind,
                        "actor_id": e.actor_id,
                        "payload": dict(e.payload),
                    }
                    for e in trace.events
                ],
            },
            default=str,
        )

    def read_baseline() -> str:
        """Return the baseline harness trajectory."""
        return _render(baseline)

    def read_candidate() -> str:
        """Return the candidate harness trajectory."""
        return _render(candidate)

    def submit_preference(score: float, rationale: str = "") -> str:
        """Finalize the signed preference. Call exactly once."""
        plan["score"] = score
        plan["rationale"] = rationale
        return json.dumps({"status": "ok"})

    return {
        "get_task": get_task,
        "read_baseline": read_baseline,
        "read_candidate": read_candidate,
        "submit_preference": submit_preference,
    }


@dataclass(slots=True)
class PreferenceJudge:
    """Interface B signed pairwise judge, shared by RHO and genetic stages."""

    agent_factory: Callable[[dict, str], str] | None = None
    app_names: Mapping[str, str] = field(default_factory=lambda: dict(APP_NAMES))

    def compare(
        self,
        task: EvolutionTask,
        baseline: ExecutionTrace,
        candidate: ExecutionTrace,
    ) -> PreferenceVerdict:
        """Compare two trajectories, returning an unavailable verdict on failure."""
        expected = _expected_answer(task)
        gt_available = bool(expected)
        plan: dict = {}
        callables = _build_callables(task, baseline, candidate, expected, plan)
        prompt = _PROMPT.format(
            gt_guidance=_GT_PRESENT if gt_available else _GT_ABSENT
        )

        run = run_workspace_agent(
            callables,
            prompt,
            app_names=self.app_names,
            agent_factory=self.agent_factory,
        )
        if not run.ok:
            return PreferenceVerdict(
                task_id=task.task_id, gt_available=gt_available, error=run.error
            )
        if "score" not in plan:
            return PreferenceVerdict(
                task_id=task.task_id,
                gt_available=gt_available,
                error="never called submit_preference; comparison discarded",
            )

        raw = plan["score"]
        try:
            score = float(raw)
        except (TypeError, ValueError):
            return PreferenceVerdict(
                task_id=task.task_id,
                gt_available=gt_available,
                error=f"score is not a number: {raw!r}",
            )
        if not MIN_SCORE <= score <= MAX_SCORE:
            return PreferenceVerdict(
                task_id=task.task_id,
                gt_available=gt_available,
                error=f"score {score} outside [{MIN_SCORE}, {MAX_SCORE}]",
            )

        if score > 0:
            winner = "candidate"
        elif score < 0:
            winner = "baseline"
        else:
            winner = "tie"

        return PreferenceVerdict(
            task_id=task.task_id,
            score=score,
            winner=winner,
            rationale=str(plan.get("rationale") or ""),
            gt_available=gt_available,
            available=True,
        )
```

```python
"""Two-level group/rollout concurrency with a global cap.

RHO's rollout phase is inherently grouped: ``k`` tasks x ``G`` rollouts, where a
group's diagnosis cannot start until all ``G`` of its rollouts finish. A single
flat worker count cannot express this -- with a flat cap of 6 the scheduler cannot
distinguish "6 tasks x 1 rollout" from "2 tasks x 3 rollouts".

The model is taken from the previous Gaia RHO runner
(``reference/evolve_run.py:99-105``):

  group_workers    concurrently admitted task groups
  rollout_workers  concurrent rollouts within one group
  global_cap       hard ceiling on simultaneous executions

and its invariant (``reference/evolve_run.py:194``):

  global_cap <= group_workers * rollout_workers

A global cap larger than the structure can produce is a configuration error, not
something to silently clamp.

Group-major admission: completing whole groups early is preferred over spreading
thinly across all ``k`` tasks, so ``on_group_complete`` can dispatch diagnosis for
a finished group while later groups are still executing.

This module is agent-neutral. It schedules opaque callables and knows nothing
about CUGA, rollouts, or traces.
"""
from __future__ import annotations

import threading
from concurrent.futures import ThreadPoolExecutor, as_completed
from dataclasses import dataclass
from typing import Callable, Sequence, TypeVar

T = TypeVar("T")


@dataclass(frozen=True, slots=True)
class ConcurrencyPlan:
    """A validated two-level concurrency configuration."""

    group_workers: int
    rollout_workers: int
    global_cap: int

    @classmethod
    def validated(
        cls, group_workers: int, rollout_workers: int, global_cap: int
    ) -> "ConcurrencyPlan":
        """Build a plan, refusing an incoherent configuration."""
        for name, value in (
            ("group_workers", group_workers),
            ("rollout_workers", rollout_workers),
            ("global_cap", global_cap),
        ):
            if isinstance(value, bool) or not isinstance(value, int) or value < 1:
                raise ValueError(f"{name} must be a positive integer, got {value!r}")
        if global_cap > group_workers * rollout_workers:
            raise ValueError(
                f"global cap {global_cap} exceeds what the structure can produce "
                f"({group_workers} groups x {rollout_workers} rollouts = "
                f"{group_workers * rollout_workers}); lower the cap or raise the "
                "worker counts"
            )
        return cls(
            group_workers=group_workers,
            rollout_workers=rollout_workers,
            global_cap=global_cap,
        )


@dataclass(frozen=True, slots=True)
class GroupResult:
    """One group's successful outcomes plus how many items failed."""

    group_id: str
    outcomes: tuple = ()
    failures: int = 0


def run_groups(
    groups: Sequence[tuple[str, Sequence[T]]],
    run_one: Callable[[str, T], object],
    plan: ConcurrencyPlan,
    on_group_complete: Callable[[GroupResult], None] | None = None,
) -> tuple[GroupResult, ...]:
    """Run every group's items, group-major, under the global cap.

    Results are returned in input group order regardless of completion order, so
    a caller's reporting is deterministic. An item that raises is counted as a
    failure and its siblings are preserved: one broken rollout must not discard a
    group's remaining evidence.
    """
    if not groups:
        return ()

    # The global cap is enforced by a semaphore shared across every group, so the
    # two-level structure can never oversubscribe the real resource.
    gate = threading.Semaphore(plan.global_cap)

    def run_group(group_id: str, items: Sequence[T]) -> GroupResult:
        outcomes: list[object] = []
        failures = 0

        def guarded(item: T) -> tuple[bool, object]:
            with gate:
                try:
                    return True, run_one(group_id, item)
                except Exception:  # noqa: BLE001 - a failure is data
                    return False, None

        if plan.rollout_workers == 1 or len(items) == 1:
            results = [guarded(item) for item in items]
        else:
            workers = min(plan.rollout_workers, len(items))
            with ThreadPoolExecutor(
                max_workers=workers, thread_name_prefix=f"rho-{group_id}"
            ) as pool:
                # map preserves input order regardless of completion order.
                results = list(pool.map(guarded, items))

        for ok, value in results:
            if ok:
                outcomes.append(value)
            else:
                failures += 1

        result = GroupResult(
            group_id=group_id, outcomes=tuple(outcomes), failures=failures
        )
        if on_group_complete is not None:
            on_group_complete(result)
        return result

    if plan.group_workers == 1 or len(groups) == 1:
        return tuple(run_group(gid, items) for gid, items in groups)

    order = {gid: i for i, (gid, _) in enumerate(groups)}
    collected: list[GroupResult] = []
    workers = min(plan.group_workers, len(groups))
    with ThreadPoolExecutor(
        max_workers=workers, thread_name_prefix="rho-group"
    ) as pool:
        futures = [
            pool.submit(run_group, gid, items) for gid, items in groups
        ]
        # as_completed so on_group_complete fires as groups finish, not at the end.
        for future in as_completed(futures):
            collected.append(future.result())
    collected.sort(key=lambda r: order[r.group_id])
    return tuple(collected)
```

### Task 11: SKIPPED — entropy floor is met by rollout count instead

**Status: not implemented. Superseded by a better fix on 2026-08-17.**

The original plan was to delete the `"skip"` tier from `EntropyTracker.classify`
so no cell was filtered out. That was the wrong repair: the evidence floor exists
because a mean built from one rollout is untrustworthy, and deleting the guard
hides that rather than fixing it.

The floor is instead satisfied **naturally** by giving each RHO candidate
`R = 2` rollouts per coreset task (`--rho-candidate-rollouts`, Tasks 13-14):

| Candidate | rollouts per coreset task | clears `min_rollouts_per_candidate=2` |
| --- | --- | --- |
| base | `G` = 3 | yes |
| candidate 0..N-1 | `R` = 2 | yes |

Comparable candidates per cell = base + N = **4**, which clears
`min_comparable_candidates=3`. So `classify` never returns `"skip"` for a cell
that RHO populated, and `EntropyTracker` keeps its guard for cells that genuinely
lack evidence.

This is informative rather than wasteful because CUGA rollouts are stochastic:
the tiny5 run produced 3/5 and then 1/5 on the same harness and tasks, a 40pp
spread. Two rollouts of one candidate really do differ.

`EntropyTracker` and `core/entropy.py` are therefore **left unchanged** by this
plan. The between/within variance decomposition remains recorded in spec section
6.1 as future work.

**One wiring requirement, pinned by a test in Task 13:**
`EntropyTracker._comparable_candidates` counts only candidates promoted through
`mark_comparable()`. Rollout count alone is not sufficient — without that call
`comp` is empty and the floor fails no matter how many rollouts were spent. The
orchestrator's own `_cell_entropy` path does not share this requirement because
it reads `pool.all_entries()` directly, so the two paths must both be satisfied.

```python
"""Post-hoc detection of ground-truth literals inside evolved artifacts.

Ground truth reaches the preference judge and the editor by explicit decision
(spec section 7), overriding the ``AGENTS.md`` no-labels rule. Containment is by
prompting, which means nothing in the data path prevents a candidate from
improving its score by carrying an answer in an artifact rather than by improving
procedure.

This module changes none of that. It runs AFTER a run and reports which exported
artifacts contain literals drawn from the dataset's answer keys, so a contaminated
harness is discovered here rather than by a reviewer at the worst possible moment.

Confidence is reported, not decided: a long distinctive phrase appearing in an
artifact is almost certainly memorization, whereas ``17`` legitimately appears in
ordinary prose.
"""
from __future__ import annotations

import re
from dataclasses import dataclass
from typing import Mapping, Sequence

#: Regexes that are placeholders rather than answer keys.
_PLACEHOLDERS = frozenset({r"(?i)\?", r"\?", r"(?i)\\?"})

#: At or above this length, a matched literal is distinctive enough that an
#: incidental occurrence is implausible.
HIGH_CONFIDENCE_MIN_LENGTH = 8


@dataclass(frozen=True, slots=True)
class ContaminationHit:
    """One ground-truth literal found inside one artifact."""

    artifact_id: str
    literal: str
    confidence: str


def literals_from_regexes(regexes: Sequence[str]) -> tuple[str, ...]:
    """Recover plain literals from ``\\b``-delimited answer-key regexes.

    The dataset stores answers as regexes like ``(?i)\\b0\\.1777\\b``. Stripping
    the inline flags, word boundaries, and backslash escapes recovers the literal
    a memorizing artifact would contain.
    """
    literals: list[str] = []
    for raw in regexes:
        if raw in _PLACEHOLDERS:
            continue
        text = raw
        text = re.sub(r"^\(\?[a-zA-Z]+\)", "", text)  # inline flags
        text = text.replace(r"\b", "")
        text = re.sub(r"\\(.)", r"\1", text)  # \. -> .  and  \  -> space
        text = text.strip()
        if not text or text == "?":
            continue
        literals.append(text)
    return tuple(dict.fromkeys(literals))


def scan_artifacts(
    artifacts: Mapping[str, str], literals: Sequence[str]
) -> tuple[ContaminationHit, ...]:
    """Report every (artifact, literal) co-occurrence, case-insensitively."""
    if not literals:
        return ()
    hits: list[ContaminationHit] = []
    for artifact_id in sorted(artifacts):
        haystack = artifacts[artifact_id].lower()
        for literal in literals:
            needle = literal.lower().strip()
            if not needle or needle not in haystack:
                continue
            hits.append(
                ContaminationHit(
                    artifact_id=artifact_id,
                    literal=literal,
                    confidence=(
                        "high"
                        if len(needle) >= HIGH_CONFIDENCE_MIN_LENGTH
                        else "low"
                    ),
                )
            )
    return tuple(hits)
```

```python
"""RHO round configuration, phase sequencing, and per-round summaries.

One outer loop; the mode selects which phases run per outer iteration:

  rho           the 10 RHO phases, repeated ``rounds`` times
  genetic       the existing mutation/crossover loop, unchanged
  rho-genetic   [RHO round -> genetic iterations] x ``rounds``

Pool, export, run logging, budgets, and the adapter are shared across all three.
This module owns only the sequencing contract and its reporting, so it stays
agent-neutral and independently testable; the live wiring lives in the pipeline.
"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Literal, Mapping

from agent_evolve.core.rho.scheduler import ConcurrencyPlan

RhoMode = Literal["rho", "genetic", "rho-genetic"]

_RHO_PHASES: tuple[str, ...] = (
    "history_load",
    "trajectory_comprehension",
    "difficulty_fingerprint",
    "coreset_selection",
    "group_rollouts",
    "group_diagnosis",
    "candidate_proposal",
    "candidate_rollouts",
    "preference_judging",
    "pool_commit",
)

PHASES: dict[str, tuple[str, ...]] = {
    "rho": _RHO_PHASES,
    "genetic": ("genetic_iterations",),
    "rho-genetic": _RHO_PHASES + ("genetic_iterations",),
}


def phases_for(mode: str) -> tuple[str, ...]:
    """Return the ordered phase sequence for ``mode``."""
    try:
        return PHASES[mode]
    except KeyError:
        raise ValueError(
            f"unknown mode: {mode!r}; expected one of {sorted(PHASES)}"
        ) from None


@dataclass(frozen=True, slots=True)
class RoundConfig:
    """Validated RHO round configuration.

    Defaults at the call site follow the paper: k=10, G=3, N=3.
    """

    mode: RhoMode
    rounds: int
    coreset_size: int
    group_rollouts: int
    candidates: int
    #: R: rollouts per candidate per coreset task. Default 2 so each candidate's
    #: mean clears EntropyTracker.min_rollouts_per_candidate=2 naturally, without
    #: weakening the evidence floor. See the Task 11 note.
    candidate_rollouts: int
    selector: str
    genetic_iterations_per_round: int
    concurrency: ConcurrencyPlan

    def __post_init__(self) -> None:
        if self.mode not in PHASES:
            raise ValueError(f"unknown mode: {self.mode!r}")
        for name in (
            "rounds",
            "coreset_size",
            "group_rollouts",
            "candidates",
            "candidate_rollouts",
        ):
            value = getattr(self, name)
            if isinstance(value, bool) or not isinstance(value, int) or value < 1:
                raise ValueError(f"{name} must be a positive integer, got {value!r}")
        if self.genetic_iterations_per_round < 0:
            raise ValueError("genetic_iterations_per_round must be >= 0")

    @property
    def rollouts_per_round(self) -> int:
        """``k*G`` baseline plus ``k*N*R`` candidate rollouts.

        At k=10, G=3, N=3, R=2 that is 30 + 60 = 90 rollouts per round. R is the
        price of clearing the entropy evidence floor honestly rather than by
        deleting the guard.
        """
        return self.coreset_size * (
            self.group_rollouts + self.candidates * self.candidate_rollouts
        )

    @property
    def phases(self) -> tuple[str, ...]:
        return phases_for(self.mode)


@dataclass(frozen=True, slots=True)
class RoundSummary:
    """One round's reportable outcome."""

    round_index: int
    selection_method: str
    coreset_ids: tuple[str, ...]
    candidates_requested: int
    candidates_distinct: int
    discarded: tuple[tuple[int, str], ...]
    cache_hits: Mapping[str, int]
    pool_size: int

    @property
    def collapsed(self) -> bool:
        """True when fewer distinct candidates survived than were requested.

        Surfaced because a collapse to one candidate means the pairwise judge is
        comparing a harness against itself, which must never be silent.
        """
        return self.candidates_distinct < self.candidates_requested

    def line(self) -> str:
        return (
            f"round {self.round_index}: {len(self.coreset_ids)} coreset tasks "
            f"({self.selection_method}), candidates "
            f"{self.candidates_distinct} of {self.candidates_requested} distinct, "
            f"pool {self.pool_size}"
        )
```

```python
def resolve_rho_config(args) -> "RoundConfig":
    """Validate RHO configuration before anything expensive is constructed.

    Both refusals here are credential-independent on purpose. A run configured
    with an impossible concurrency is wrong whether or not a model is reachable,
    and reporting "no model configured" first would send an operator to fix the
    wrong thing.
    """
    from agent_evolve.core.rho.rounds import RoundConfig
    from agent_evolve.core.rho.scheduler import ConcurrencyPlan

    if (
        args.rho_proposal_temperature is not None
        and args.rho_proposal_temperature == 0.0
    ):
        raise SystemExit(
            "--rho-proposal-temperature 0.0 is rejected by the endpoint "
            "('temperature does not support 0.0'); omit the flag instead"
        )
    try:
        plan = ConcurrencyPlan.validated(
            group_workers=args.rho_group_workers,
            rollout_workers=args.rho_rollout_workers,
            global_cap=args.max_workers,
        )
    except ValueError as exc:
        raise SystemExit(f"invalid RHO concurrency: {exc}") from exc

    try:
        return RoundConfig(
            mode=args.mode,
            rounds=args.rho_rounds,
            coreset_size=args.rho_coreset_size,
            group_rollouts=args.rho_group_rollouts,
            candidates=args.rho_candidates,
            candidate_rollouts=args.rho_candidate_rollouts,
            selector=args.rho_selector,
            genetic_iterations_per_round=args.genetic_iterations_per_round,
            concurrency=plan,
        )
    except ValueError as exc:
        raise SystemExit(f"invalid RHO configuration: {exc}") from exc
```
